4. **Add code-behind logic**:
```csharp
private bool _isGenerating;
private string? _validationError;

private async Task OnGenerateClicked()
{
    // Inline validation
    _validationError = null;
    if (string.IsNullOrWhiteSpace(_config.ProjectName))
    {
        _validationError = "Project name is required";
        return;
    }
    if (_config.ProjectName.Any(c => !char.IsLetterOrDigit(c) && c != '.' && c != '-' && c != '_'))
    {
        _validationError = "Project name can only contain letters, digits, dots, hyphens, and underscores";
        return;
    }

    _isGenerating = true;
    StateHasChanged();
    await Task.Yield(); // Let spinner render before CPU work

    try
    {
        var files = GenerationService.Generate(_config);
        using var zipStream = ZipService.CreateZip(files);
        var fileName = $"{_config.ProjectName}.zip";
        using var streamRef = new DotNetStreamReference(stream: zipStream);
        await JS.InvokeVoidAsync("downloadFileFromStream", fileName, streamRef);
    }
    finally
    {
        _isGenerating = false;
    }
}
```
